"use client";

import { useState } from "react";

export function CredentialLoginForm() {
  const [notice, setNotice] = useState(false);
  return <form className="login-dialog__credentials" noValidate onSubmit={(event) => { event.preventDefault(); setNotice(true); }}>
    <label htmlFor="login-username">Tên đăng nhập</label>
    <input type="text" id="login-username" autoComplete="username" />
    <label htmlFor="login-password">Mật khẩu</label>
    <input type="password" id="login-password" autoComplete="current-password" />
    <button className="login-dialog__submit" type="submit">Đăng nhập</button>
    {notice && <p className="login-dialog__status" role="status" aria-live="polite">Đăng nhập bằng tài khoản sắp ra mắt. Vui lòng dùng Google.</p>}
  </form>;
}
